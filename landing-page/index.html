Ejemplo práctico de transformación de un código vulnerable Python a una arquitectura segura y probada.

user_service.py — DevSecOps Code Review
Entrada (Código Inseguro) SQLi + Monolítico 
# ❌ VULNERABLE & MONOLÍTICO
import sqlite3

class UserService:
    def login(self, username, password):
        # 🚨 Vulnerabilidad SQL Injection (OWASP A03)
        query = f"SELECT * FROM users WHERE user='{username}' AND pass='{password}'"
        conn = sqlite3.connect('app.db')
        cursor = conn.cursor()
        cursor.execute(query) # Inyección directa!
        
        # 🚨 Monolito (Violación SRP - loguea y envía email)
        print(f"Log: Usuario {username} ingresó")
        self.send_email(username)
        return cursor.fetchone()
Salida Refactorizada (Gemini Skill) Prepared Query + SOLID 
# ✅ SEGURO & SOLID (DIP / SRP)
from abc import ABC, abstractmethod

class UserRepository(ABC):
    @abstractmethod
    def find_by_credentials(self, username, password): pass

class SqliteUserRepo(UserRepository):
    def find_by_credentials(self, user, pwd):
        #🔒 Consulta Parametrizada (Previene SQLi)
        query = "SELECT id, user FROM users WHERE user=? AND pass=?"
        return self.db.execute(query, (user, pwd)).fetchone()
